cmake_minimum_required(VERSION 3.24)
project(NFSMWUnlimiterCustomization LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure on Windows with -A Win32; NFSMW 1.3 is x86.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(NFSMWUnlimiter SHARED dllmain.cpp NFSMWUnlimiter.rc)
target_include_directories(NFSMWUnlimiter PRIVATE includes)
target_compile_definitions(NFSMWUnlimiter PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(NFSMWUnlimiter PRIVATE /W3 /utf-8 /EHsc /Zc:strictStrings- /permissive)
target_link_libraries(NFSMWUnlimiter PRIVATE minhook bcrypt)
set_target_properties(NFSMWUnlimiter PROPERTIES PREFIX "" SUFFIX ".asi")
enable_testing()
add_executable(customization_math_tests tests/customization_math_tests.cpp)
target_compile_options(customization_math_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME customization_math COMMAND customization_math_tests)
add_executable(customization_save_tests tests/customization_save_tests.cpp)
target_link_libraries(customization_save_tests PRIVATE minhook bcrypt)
target_compile_definitions(customization_save_tests PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(customization_save_tests PRIVATE /W4 /utf-8 /EHsc)
target_link_options(customization_save_tests PRIVATE /DYNAMICBASE:NO /BASE:0x400000)
add_test(NAME customization_save COMMAND customization_save_tests)
add_executable(customization_control_state_tests tests/customization_control_state_tests.cpp)
target_compile_options(customization_control_state_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME customization_control_state COMMAND customization_control_state_tests)
add_executable(customization_controls_save_tests tests/customization_controls_save_tests.cpp)
target_link_libraries(customization_controls_save_tests PRIVATE minhook bcrypt)
target_compile_definitions(customization_controls_save_tests PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(customization_controls_save_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME customization_controls_save COMMAND customization_controls_save_tests)

add_executable(customization_profile_host_tests tests/customization_profile_host_tests.cpp)
